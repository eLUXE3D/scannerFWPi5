#****FOR X3D SCAN****

#KEYGEN - to sign pc software using inno setup
#Run powerShell as ADMIN
#note need to use -NotAfter, else default is to expire after 1 year.
#default NotBefore time is 10 minutes.

New-SelfSignedCertificate -Type Custom -Subject "Tupel" -KeyUsage DigitalSignature -FriendlyName "TupelKey" -CertStoreLocation "Cert:\CurrentUser\My" -TextExtension @("2.5.29.37={text}1.3.6.1.5.5.7.3.3", "2.5.29.19={text}") -NotAfter (Get-Date).AddYears(500)

Set-Location Cert:\CurrentUser\My
Get-ChildItem | Format-Table Subject, FriendlyName, Thumbprint

#replace code B6BE... with serial number of key from windows power shell list.

$pwd = ConvertTo-SecureString -String Turing1945x -Force -AsPlainText 
Export-PfxCertificate -cert "Cert:\CurrentUser\My\0830ED8EB0D1310CA01DDCB1503AD2F753D58B0D" -FilePath C:\Work-TupelGit\Teeth\keys\windowsKeyX3D.pfx -Password $pwd


#SignTool in innosetup
#go to tools, configure signtools
#put name as signtoolX3D
#put command as:

"C:\Program Files (x86)\Windows Kits\10\App Certification Kit\signtool.exe" sign /f "C:\Work-TupelGit\Teeth\keys\windowsKeyX3D.pfx" /t http://timestamp.comodoca.com/authenticode /p Turing1945x $f

#in [setup]
SignTool=signtoolX3D

Change AppId to a new value so it does not install over the other versions.
